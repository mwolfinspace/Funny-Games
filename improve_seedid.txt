/* === BASE-81 CODEC === */
const Base81 = {
    // 10 + 26 + 26 + 19 = 81. Excludes URL-sensitive chars like space, %, &, ?, #, etc.
    chars: "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ!#$%&()*+,-./:;<=>?@[]^",
    
    // Encode positive BigInt to Base81 string (most significant char first)
    encode: (num) => {
        if (num === 0n) return Base81.chars[0];
        let result = "";
        const base = 81n;
        while (num > 0n) {
            result = Base81.chars[Number(num % base)] + result;
            num /= base;
        }
        return result;
    },
    
    // Decode Base81 string to BigInt
    decode: (str) => {
        let num = 0n;
        const base = 81n;
        for (let i = 0; i < str.length; i++) {
            const idx = Base81.chars.indexOf(str[i]);
            if (idx === -1) throw new Error("Invalid Base81 char: " + str[i]);
            num = num * base + BigInt(idx);
        }
        return num;
    },
    
    // Encode dimensions 5-20 as single char (0-15)
    encodeDim: (size) => {
        if (size < 5 || size > 20) throw new Error("Dimension out of range (5-20)");
        return Base81.chars[size - 5]; 
    },
    
    decodeDim: (char) => {
        const idx = Base81.chars.indexOf(char);
        if (idx < 0 || idx > 15) return null; // Invalid dimension marker
        return idx + 5;
    }
};

/* === STABLE PRNG (Mulberry32) === */
// Better than Math.sin: 32-bit state, high quality, consistent across JS engines
const PRNG = {
    fromString: (str) => {
        // Hash string to 32-bit unsigned seed
        let h = 1779033703 ^ str.length;
        for(let i = 0; i < str.length; i++) {
            h = Math.imul(h ^ str.charCodeAt(i), 3432918353);
            h = h << 13 | h >>> 19;
        }
        return () => {
            h = Math.imul(h ^ (h >>> 16), 2246822507);
            h = Math.imul(h ^ (h >>> 13), 3266489909);
            return (h >>> 0) / 4294967296;
        };
    },
    
    fromSeed: (seed) => {
        // seed is integer 0 to 2^32-1
        let t = seed >>> 0;
        return () => {
            t += 0x6D2B79F5;
            let r = Math.imul(t ^ (t >>> 15), t | 1);
            r ^= r + Math.imul(r ^ (r >>> 7), r | 61);
            return ((r ^ (r >>> 14)) >>> 0) / 4294967296;
        };
    }
};

/* === SEED FORMATS === */
const SeedCodec = {
    // Short Seed: [W][H][Random(5 chars)] → ~7 chars total
    // Generates grid via PRNG. Not 100% collision-proof but 3.5 billion combinations.
    createShort: (w, h) => {
        const rngSeed = Math.floor(Math.random() * 4294967296); // 32-bit
        const seedPart = Base81.encode(BigInt(rngSeed)).padStart(5, Base81.chars[0]); // Fixed 5 chars
        return Base81.encodeDim(w) + Base81.encodeDim(h) + seedPart;
    },
    
    // Long Seed: [W][H][GridBitmap] → variable length (10-70 chars)
    // Encodes exact solution. Stable forever regardless of RNG changes.
    createLong: (grid, w, h) => {
        // Flatten to bit string
        let bits = "";
        for (let y = 0; y < h; y++) {
            for (let x = 0; x < w; x++) {
                bits += grid[y][x] ? "1" : "0";
            }
        }
        // Pad to multiple of 6 bits for clean Base81? Not needed with BigInt boundary.
        const num = BigInt("0b" + bits);
        const data = Base81.encode(num);
        return Base81.encodeDim(w) + Base81.encodeDim(h) + data;
    },
    
    // Decode any format. Returns: { type: 'short'|'long', w, h, seed?, grid? }
    decode: (str) => {
        if (str.length < 2) return null;
        
        const w = Base81.decodeDim(str[0]);
        const h = Base81.decodeDim(str[1]);
        
        // If first two chars aren't valid dimensions, assume LEGACY format (old hash-based)
        if (w === null || h === null) return { type: 'legacy', data: str };
        
        const payload = str.substring(2);
        if (payload.length <= 6 && !payload.match(/^[0]/) && payload.length > 0) {
            // Likely Short Seed (5-6 chars randomness)
            // Reconstruct 32-bit seed (may have leading zeros trimmed in encoding)
            const num = Base81.decode(payload.padStart(5, Base81.chars[0]));
            return { type: 'short', w, h, seed: Number(num) };
        } else {
            // Long Seed (exact grid)
            const num = Base81.decode(payload);
            let bits = num.toString(2).padStart(w * h, '0');
            const grid = [];
            let ptr = 0;
            for (let y = 0; y < h; y++) {
                const row = [];
                for (let x = 0; x < w; x++) {
                    row.push(bits[ptr++] === '1' ? 1 : 0);
                }
                grid.push(row);
            }
            return { type: 'long', w, h, grid };
        }
    }
};

/* === UPDATED GENERATOR === */
const Gen = {
    // Main entry point. Seed can be old style ("10xabc"), new short ("aa!7x9"), or new long ("aa91z8...")
    getGrid: (seedStr, fallbackW, fallbackH) => {
        const parsed = SeedCodec.decode(seedStr);
        
        if (parsed.type === 'legacy') {
            // Fall back to old hash-based generation for backward compatibility
            return Gen.getGridLegacy(seedStr, fallbackW, fallbackH);
        }
        
        const w = parsed.w;
        const h = parsed.h;
        let rng;
        
        if (parsed.type === 'short') {
            // Deterministic from 32-bit seed
            rng = PRNG.fromSeed(parsed.seed);
        } else {
            // Long seed: return exact grid, ignore RNG entirely
            return parsed.grid;
        }
        
        // Generate random grid with clustering (same logic as before but stable RNG)
        const g = Array(h).fill(0).map(() => Array(w).fill(0));
        for (let y = 0; y < h; y++) {
            for (let x = 0; x < w; x++) {
                let v = rng() > 0.55 ? 1 : 0;
                if (x > 0 && g[y][x - 1]) v = rng() > 0.35 ? 1 : 0;
                if (y > 0 && g[y - 1][x]) v = rng() > 0.35 ? 1 : 0;
                g[y][x] = v;
            }
        }
        // Ensure at least one cell filled
        if (g.every(r => r.every(c => c === 0))) g[Math.floor(h/2)][Math.floor(w/2)] = 1;
        return g;
    },
    
    // Legacy hash generator (keep for old seeds)
    getGridLegacy: (s, w, h) => {
        let S = typeof s === 'string' ? Gen.hash(s) : s;
        const rng = () => { const x = Math.sin(S++) * 10000; return x - Math.floor(x); };
        let g = Array(h).fill(0).map(() => Array(w).fill(0));
        for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
            let v = rng() > 0.55 ? 1 : 0;
            if (x > 0 && g[y][x - 1]) v = rng() > 0.35 ? 1 : 0;
            if (y > 0 && g[y - 1][x]) v = rng() > 0.35 ? 1 : 0;
            g[y][x] = v;
        }
        if (g.every(r => r.every(c => c === 0))) g[h >> 1][w >> 1] = 1;
        return g;
    },
    
    hash: (s) => { let h = 5381; for (let i = 0; i < s.length; i++)h = ((h << 5) + h) + s.charCodeAt(i); return Math.abs(h) },
    getClues: (g) => { 
        const l = a => { let r = [], c = 0; for (let v of a) { if (v) c++; else { if (c) r.push(c); c = 0 } } if (c) r.push(c); return r.length ? r : [0] }; 
        return { rows: g.map(l), cols: g[0].map((_, i) => l(g.map(r => r[i]))) } 
    }
};

const Base81 = {
    chars: "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ!#$%&()*+,-./:;<=>?@[]^",
    encode: (num, pad=0) => {
        if (num === 0n) return Base81.chars[0].repeat(pad || 1);
        let res = "";
        const base = 81n;
        while (num > 0n) {
            res = Base81.chars[Number(num % base)] + res;
            num /= base;
        }
        return res.padStart(pad, Base81.chars[0]);
    },
    decode: (str) => {
        let num = 0n;
        for (let c of str) num = num * 81n + BigInt(Base81.chars.indexOf(c));
        return num;
    },
    dim: {
        enc: (n) => Base81.chars[n-5], // 5->0, 20->15
        dec: (c) => { const i=Base81.chars.indexOf(c); return (i>=0&&i<=15)?i+5:null; }
    }
};

/* === CLUE-COMPRESSED EDITOR CODEC === */
const EditorCodec = {
    // Pack clues efficiently. Average 10x10 puzzle: ~15 chars vs ~40 raw.
    packClues: (rows, cols) => {
        // Flatten all numbers 0-20 into single chars (0=sep, 1-20=values)
        // 21 values fit in 0-20, but we have 81 chars available.
        // We use 0-19 for values 1-20, 20 as row separator, 21+ as multi-byte lengths (rarely needed)
        let out = "";
        for (let row of rows) {
            if (row.length === 1 && row[0] === 0) out += Base81.chars[0]; // Empty row marker
            else for (let v of row) out += Base81.chars[v]; // Values 1-20 map to chars 1-20 (0 reserved for sep)
            out += Base81.chars[20]; // Row separator
        }
        out += Base81.chars[20]; // Double sep = start cols
        for (let col of cols) {
            if (col.length === 1 && col[0] === 0) out += Base81.chars[0];
            else for (let v of col) out += Base81.chars[v];
            out += Base81.chars[20];
        }
        return out;
    },
    
    unpackClues: (str) => {
        const parts = str.split(Base81.chars[20]);
        const rows = [], cols = [];
        let mode = 'rows';
        for (let part of parts) {
            if (!part && mode === 'rows') { mode = 'cols'; continue; } // Double sep
            if (!part) continue; // trailing
            const clues = [];
            for (let c of part) {
                const v = Base81.chars.indexOf(c);
                clues.push(v === 0 ? 0 : v); // 0->0(empty), 1->1, 20->20 etc
            }
            if (clues.length === 1 && clues[0] === 0) (mode==='rows'?rows:cols).push([0]);
            else (mode==='rows'?rows:cols).push(clues);
        }
        return {rows, cols};
    },

    // MAIN: Try to encode as clues (short), fallback to bitmap (long)
    encode: (grid, w, h) => {
        // Check if line-solvable (deterministic)
        const check = Solver.check(grid);
        if (check.c === "ok") {
            // Can reconstruct from clues! Store only clues.
            const clues = Gen.getClues(grid);
            const packed = EditorCodec.packClues(clues.rows, clues.cols);
            return "C" + Base81.dim.enc(w) + Base81.dim.enc(h) + packed; // C = Clue mode
        } else {
            // Ambiguous art: store full bitmap with RLE
            return "G" + EditorCodec.encodeBitmap(grid, w, h); // G = Grid mode
        }
    },

    decode: (str) => {
        if (str[0] === 'C') {
            // Clue mode: Reconstruct via solver
            const w = Base81.dim.dec(str[1]);
            const h = Base81.dim.dec(str[2]);
            const {rows, cols} = EditorCodec.unpackClues(str.slice(3));
            // Reconstruct using solver (must be line-solvable)
            return EditorCodec.solveFromClues(w, h, rows, cols);
        } else if (str[0] === 'G') {
            // Grid mode
            return EditorCodec.decodeBitmap(str.slice(1));
        }
        return null;
    },

    // Bitmap encoding with RLE for long runs (fallback for ambiguous art)
    encodeBitmap: (grid, w, h) => {
        let bits = grid.flat().join('');
        // Simple RLE: count consecutive same bits, alternate starting with 0s (padding if needed)
        let runs = [];
        let curr = bits[0] === '1' ? 1 : 0;
        if (bits[0] === '1') runs.push(0); // Lead with 0-length white if starts with black
        let count = 0;
        for (let b of bits) {
            if (parseInt(b) === curr) count++;
            else { runs.push(count); curr = 1-curr; count = 1; }
        }
        runs.push(count);
        
        // Pack into Base-81: Use 80 as continuation for runs > 80
        let out = Base81.dim.enc(w) + Base81.dim.enc(h);
        let i = 0;
        while (i < runs.length) {
            let r = runs[i];
            while (r > 80) { out += Base81.chars[80]; r -= 80; }
            out += Base81.chars[r];
            i++;
        }
        return out;
    },

    // Reconstruct exact grid from clues using deterministic solver
    solveFromClues: (w, h, rowClues, colClues) => {
        // Use the game's existing solver to fill grid
        let grid = Array(h).fill().map(()=>Array(w).fill(0));
